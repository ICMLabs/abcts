## The baseline RATIO script — the flip's discovery mechanism

Snapshot the baselines BEFORE the flip, re-record after, then run this. It prints one row
per baseline row KIND with how many numbers failed to scale by 7.75 and what ratios they
came out at — an exhaustive, judgement-free list of what still holds a staff space.

```bash
cp -r tests/renderer/baselines /tmp/baselines-before
# … flip UNIT_PX to 1 …
npm run baseline
python3 /tmp/ratio.py
```

```python
# /tmp/ratio.py
import re, os
old='/tmp/baselines-before'; new='tests/renderer/baselines'
num=re.compile(r'-?\d+\.\d+|-?\d+')
bad={}
for f in sorted(os.listdir(old)):
    if not f.endswith('.txt'): continue
    A=open(os.path.join(old,f)).read().splitlines()
    B=open(os.path.join(new,f)).read().splitlines()
    if len(A)!=len(B):
        print('LINE COUNT DIFFERS', f, len(A), len(B)); continue
    for i,(a,b) in enumerate(zip(A,B)):
        na=[float(x) for x in num.findall(a)]
        nb=[float(x) for x in num.findall(b)]
        if len(na)!=len(nb): print('TOKEN COUNT', f, i, a, b); continue
        for x,y in zip(na,nb):
            if x==0 and y==0: continue
            if x==0 or y==0: continue
            if abs(y/x-7.75)>0.02:
                k=a.strip().split()[0]
                e=bad.setdefault(k,[0,set()]); e[0]+=1; e[1].add(round(y/x,3))
for k,(n,rs) in sorted(bad.items(), key=lambda kv:-kv[1][0]):
    print(f'{k:16} {n:6}  ratios {sorted(rs)[:6]}')
```

**A row kind at zero is a subsystem already converted.** At the point of revert, `staffline`
and every glyph placement were clean; `note@N` widths and `staff` originY were not.

## The DOM-contract probe, `/tmp/domprobe.mjs`

Prints OUR contract beside abcjs's, row by row, with an `X` on each mismatch. It is what
turned each of this session's nine DOM findings into a one-run question.

```js
import { readFileSync } from 'node:fs'
import { renderAbc } from './src/compat/index.ts'
const slug = process.argv[2]
const j = JSON.parse(readFileSync(`tests/corpus-dom/${slug}.json`, 'utf-8'))
function contractOf(svg) {
  const rows = []; const stack = []
  const depth = () => stack.filter(Boolean).length
  for (const m of svg.matchAll(/<(\/?)([a-zA-Z]+)([^>]*?)(\/?)>/g)) {
    const [, closing, tag, attrs, selfClose] = m
    if (tag==='svg'||tag==='style'||tag==='title'||tag==='defs') {
      if (closing==='/'&&tag!=='svg') stack.pop()
      else if (closing===''&&selfClose===''&&tag!=='svg') stack.push(false)
      continue
    }
    if (closing==='/') { stack.pop(); continue }
    const klass = /\sclass="([^"]*)"/.exec(attrs??'')?.[1] ?? null
    const name = /\sdata-name="([^"]*)"/.exec(attrs??'')?.[1] ?? null
    const counts = klass!==null||name!==null
    if (counts) rows.push({depth:depth(), class: klass===''?null:klass, name})
    if (selfClose==='') stack.push(counts)
  }
  return rows
}
const got = contractOf(renderAbc('paper', j.abc, {add_classes:true})[0].svg)
const want = j.contract
const s = r => r===undefined?'—':`d${r.depth} ${r.class??'·'} | ${r.name??'·'}`
console.log('ABC:', JSON.stringify(j.abc))
for (let i=0;i<Math.max(got.length,want.length);i++) {
  const ok = got[i]&&want[i]&&got[i].depth===want[i].depth&&got[i].class===want[i].class&&got[i].name===want[i].name
  console.log(String(i).padStart(3), ok?' ':'X', s(got[i]).padEnd(72), '||', s(want[i]))
}
```

## The height classifier, `/tmp/probe9.mjs`

Unchanged from `-08-10c`; still says `{ exact: 80, ulpOnly: 86, structural: 5 }`. It is what
keeps the ULP tail and the two real vertical defects from being read as one bug.

```js
import { readFileSync, readdirSync, existsSync } from 'node:fs'
import { renderAbc } from './src/compat/index.ts'
const dir='tests/corpus-abcjs/fixtures', gdir='tests/corpus-abcjs/golden'
const H = s => /height="([\d.]+)"/.exec(s)?.[1]
let exact=0, tiny=0, big=[]
for (const f of readdirSync(dir).filter(f=>f.endsWith('.abc')).sort()) {
  const slug=f.replace(/\.abc$/,''); if(!existsSync(`${gdir}/${slug}.svg`)) continue
  const want=H(readFileSync(`${gdir}/${slug}.svg`,'utf-8'))
  let got; try { got=H(renderAbc('paper', readFileSync(`${dir}/${f}`,'utf-8'), {staffwidth:670})[0]?.svg ?? '') } catch(e){}
  if(!want||!got) continue
  if(got===want){exact++;continue}
  const d=Math.abs(Number(got)-Number(want))
  if(d/Number(want) < 1e-12) tiny++; else big.push([slug, got, want, d.toExponential(2)])
}
console.log({exact, ulpOnly:tiny, structural:big.length})
for (const b of big) console.log(' ', b.join('  '))
```
