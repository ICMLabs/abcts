## The masked-height probe, for recreating `/tmp/probe4.mjs`

```js
import { readFileSync, readdirSync, existsSync, writeFileSync } from 'node:fs'
import { renderAbc } from './src/compat/index.ts'
const dir='tests/corpus-abcjs/fixtures', gdir='tests/corpus-abcjs/golden'
const strip = (s) => s.replace(/height="[\d.]+"/, 'height="H"')
const rows=[]
for (const f of readdirSync(dir).filter(f=>f.endsWith('.abc')).sort()) {
  const slug=f.replace(/\.abc$/,'')
  if(!existsSync(`${gdir}/${slug}.svg`)) continue
  const want=strip(readFileSync(`${gdir}/${slug}.svg`,'utf-8'))
  let got=''
  try { got=strip(renderAbc('paper', readFileSync(`${dir}/${f}`,'utf-8'), {staffwidth:670})[0]?.svg ?? '') } catch(e){}
  const n=Math.min(got.length,want.length); let i=0
  while(i<n && got[i]===want[i]) i++
  if(i===n && got.length===want.length) { rows.push({slug,i:Infinity,w:''}); continue }
  rows.push({slug,i,w:`got  …${got.slice(Math.max(0,i-40),i+70)}\nwant …${want.slice(Math.max(0,i-40),i+70)}`})
}
rows.sort((a,b)=>b.i-a.i)
writeFileSync('/tmp/probe-noheight.txt', rows.map(r=>`${r.slug.padEnd(40)} ${r.i}\n  ${r.w}`).join('\n'))
```

Run with `npx vite-node /tmp/probe4.mjs`, then histogram the families:

## The height-classifier probe, `/tmp/probe9.mjs`

```js
import { readFileSync, readdirSync, existsSync } from 'node:fs'
import { renderAbc } from './src/compat/index.ts'
const dir='tests/corpus-abcjs/fixtures', gdir='tests/corpus-abcjs/golden'
const H = s => /height="([\d.]+)"/.exec(s)?.[1]
let exact=0, tiny=0, big=[]
for (const f of readdirSync(dir).filter(f=>f.endsWith('.abc')).sort()) {
  const slug=f.replace(/\.abc$/,''); if(!existsSync(`${gdir}/${slug}.svg`)) continue
  const want=H(readFileSync(`${gdir}/${slug}.svg`,'utf-8'))
  let got; try { got=H(renderAbc('paper', readFileSync(`${dir}/${f}`,'utf-8'), {staffwidth:670})[0]?.svg ?? '') } catch(e){}
  if(!want||!got) continue
  if(got===want){exact++;continue}
  const d=Math.abs(Number(got)-Number(want))
  if(d/Number(want) < 1e-12) tiny++; else big.push([slug, got, want, d.toExponential(2)])
}
console.log({exact, ulpOnly:tiny, structural:big.length})
for (const b of big) console.log(' ', b.join('  '))
```
